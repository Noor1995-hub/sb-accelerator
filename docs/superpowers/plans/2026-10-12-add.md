> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new `/add` skill takes one item (pasted text or a file, or this conversation), asks only what it cannot tell, writes it as already accepted (a meeting note, or a hidden session record for a conversation), and processes it into project pages and decisions right away.

**Architecture:** Text reuses `/sync-recordings`' inbox worker and the #277 commands (`meeting triage-apply`, `reconcile-reviews`). Conversations get a new CLI `session write` that stores a hidden record in `_messages/sessions/`. `update-queue` learns `newSessions`, and the shared `process.md` reads them, so project pages and decisions cite `(session · …)`. `decision write`'s source lookup and `links.py` learn the subfolder. Plaud stays with `/sync-recordings`, whose chat triage gains a project suggestion for meetings with no project (Task 5).

**Tech Stack:** Python 3 stdlib (`plugin/lib/vaultlib`), pytest (`scripts/tests`), skill markdown.

**Spec:** `docs/superpowers/specs/2026-10-12-add-design.md`

```python
"""Session records (#267): what a conversation with Claude added to the brain.

A conversation is not a meeting and gets no visible note. Its summary is kept
here, in the hidden `_messages/sessions/`, so project pages and decisions can
cite where a line came from and the evening run's queue can tell it has been
read. Every mail and Teams reader globs `_messages/*-messages.md` in the folder
itself, so nothing there ever sees these files.
"""
from __future__ import annotations

import re
from datetime import date, datetime
from pathlib import Path
from typing import List, Optional

from . import markdown, naming, vault

SESSIONS_DIR = Path(vault.MESSAGES_DIR) / "sessions"
REQUIRED = ("title", "date", "projects", "people", "summary", "scope")
OPTIONAL = ("decisions",)
QUOTE_KEYS = frozenset({"title", "scope"})
_NEWLINE_RE = re.compile(r"[\r\n]")


class SessionError(Exception):
    """A payload field or vault state the caller must fix."""


def _line(value, field: str) -> str:
    text = str(value or "").strip()
    if not text:
        raise SessionError(f"{field} must not be empty")
    if _NEWLINE_RE.search(text):
        raise SessionError(f"{field} must be one line")
    return text


def _lines(value, field: str, *, allow_empty: bool) -> List[str]:
    if not isinstance(value, list) or (not value and not allow_empty):
        raise SessionError(f"{field} must be a {'' if allow_empty else 'non-empty '}list")
    return [_line(item, f"{field} entry") for item in value]


def _slugs(root: Path, value, field: str, exists) -> List[str]:
    if not isinstance(value, list):
        raise SessionError(f"{field} must be a list of slugs")
    for slug in value:
        try:
            naming.validate_slug(str(slug))
        except ValueError as exc:
            raise SessionError(f"{field} entry {slug!r}: {exc}") from exc
        if not exists(str(slug)).is_file():
            raise SessionError(f"{field} entry {slug!r} matches no note — "
                               "check the spelling")
    return [str(slug) for slug in value]


def write(root: Path, payload: dict, *, now: Optional[datetime] = None) -> dict:
    if not isinstance(payload, dict):
        raise SessionError("the payload must be a JSON object")
    unknown = sorted(set(payload) - set(REQUIRED) - set(OPTIONAL))
    if unknown:
        raise SessionError(f"unknown payload key(s) {unknown}")
    missing = [key for key in REQUIRED if key not in payload]
    if missing:
        raise SessionError(f"missing payload key(s) {missing}")
    title = _line(payload["title"], "title")
    try:
        day = date.fromisoformat(str(payload["date"])).isoformat()
    except ValueError as exc:
        raise SessionError(f"date {payload['date']!r} is not YYYY-MM-DD") from exc
    projects = _slugs(root, payload["projects"], "projects",
                      lambda s: root / vault.PROJECTS_DIR / s / "index.md")
    people = _slugs(root, payload["people"], "people",
                    lambda s: root / vault.PEOPLE_DIR / f"{s}.md")
    summary = _lines(payload["summary"], "summary", allow_empty=False)
    decisions = _lines(payload.get("decisions", []), "decisions", allow_empty=True)
    scope = _line(payload["scope"], "scope")
    stamp = (now or datetime.now()).isoformat(timespec="minutes")

    folder = root / SESSIONS_DIR
    folder.mkdir(parents=True, exist_ok=True)
    base = f"{day}-{naming.slugify(title)}"
    stem, n = base, 1
    while (folder / f"{stem}.md").exists():
        n += 1
        stem = f"{base}-{n}"

    frontmatter = {"type": "session", "date": day, "updated-at": stamp,
                   "projects": projects, "people": people, "scope": scope}
    lines = [markdown.title_heading(title), "", "## Summary"]
    lines += [f"- {item}" for item in summary]
    if decisions:
        lines += ["", "## Decisions"] + [f"- {item}" for item in decisions]
    text = markdown.emit_frontmatter(frontmatter, QUOTE_KEYS) + "\n".join(lines) + "\n"
    target = folder / f"{stem}.md"
    vault.atomic_create(target, text)
    return {"path": target.relative_to(root).as_posix(), "stem": stem}
```

If `vault.atomic_create` raises on an existing file instead of returning, keep the suffix loop as written: it already avoids the clash.

```python
def _index_sessions(root: Path) -> Tuple[List[dict], List[str]]:
    """Session records from /add (#267). Activity is `updated-at`, the write
    time, like a message log. A damaged record warns and is skipped."""
    out: List[dict] = []
    warnings: List[str] = []
    folder = root / vault.MESSAGES_DIR / "sessions"
    for path in sorted(folder.glob("*.md")) if folder.is_dir() else []:
        rel = path.relative_to(root).as_posix()
        try:
            data, _ = markdown.split_frontmatter(path.read_text(encoding="utf-8"))
        except (ValueError, OSError) as exc:
            warnings.append(f"{rel}: {exc}")
            continue
        out.append({
            "path": rel,
            "updated": _parse_iso_datetime(data.get("updated-at")),
            "projects": _slug_list(data.get("projects"), "projects", rel, warnings),
        })
    return out, warnings
```

````markdown
# Add

The owner hands over one thing. You work out what it is about and ask only
what you cannot tell; the CLI writes it as already accepted, and the brain is
updated before the owner leaves the chat.

## Requirements

Writing to the vault needs a shell, so this skill runs in **Cowork** or
**Claude Code**. It cannot run in Chat; say so plainly.

Set `VAULT` to the vault root, and:

```bash
VAULTLIB="${CLAUDE_PLUGIN_ROOT}/lib/vaultlib/cli.py"
```

Read `ok` on every call; on failure surface `error` and `hint`. Run
`python3 "$VAULTLIB" config read --vault "$VAULT"` once at the start: its
`owner` and `timezone` are used below.

## Usage

```
/add                      # this conversation, or what the owner pastes or names next
/add <text or file path>  # that text or file
```

## Rules

1. **One item per call**: one pasted text or file, or this
   one conversation. Several: call `/add` again.
2. **Nothing is written before the owner's OK** on the step-4 line.
3. **Always processed right away.** There is no "now or tonight".
4. **This skill never creates, writes or edits anything under `60-daily-notes/`.**
   Never reads mail or Teams.
5. **Never kept:** passwords, keys and tokens; anything from `95-private/`;
   personal matters about other people that are not about the work. When in
   doubt, leave it out and say so on the step-4 line.
6. **Never compute a date.** `today` is the `today` field of
   `python3 "$VAULTLIB" dates plan --vault "$VAULT" --timezone <config.timezone>`.

## Workflow

### 1. What was handed over

- **Text or a file**: pasted text, or a path.
- **This conversation**: `/add` with nothing else, or "add this conversation" /
  "add what we decided about X".

Unclear → ask once: *"This conversation, or something you'll paste?"* A
recording from a recorder or Teams → say that `/sync-recordings` fetches it
and lets you set its project, and end.

### 2. Get the content

**Text or a file.** Pasted text, or the file's text. A deck, spreadsheet,
image or other non-text file: refuse with that reason, write nothing.

**This conversation.** Summarize it yourself in step 3: the whole
conversation, or only the part the owner named.

### 3. Understand

Decide, from the content:

- **type**: `meeting` (a record of an interaction between people) or `note`
  (context, not a meeting) for a recording or text. A conversation with
  Claude is always a **session**.
- **projects**: zero or more active projects, from project names, `keywords`,
  entity names and the people involved.
- **people**: existing people by name. Someone new is created only after the
  step-4 OK.

Ask one short question per thing you cannot tell. Never more than needed.

### 4. One line, one OK

Show one line and wait for a yes:

`Meeting · Acme ERP rollout · Jan Peeters, Sofie Maes · "Forecast call moves to Thursday"`

For a session, show the line plus at most five bullets of what will be kept:

```
Session · Acme ERP rollout · "Pricing approach for Acme"
- Fixed price for phase 1
- Time and material after go-live
```

Anything you left out under rule 5: say so in one short line under it. A
correction from the owner → show the line again. No → write nothing, end.

### 5. Write

Take the run lock:

```bash
python3 "$VAULTLIB" runlock acquire --vault "$VAULT" --routine add
```

`acquired: false` → nothing written; say *"Your evening run is busy; try
again in a few minutes."* and end. **From here until the end of step 6, every
exit releases the lock first:**

```bash
python3 "$VAULTLIB" runlock release --vault "$VAULT" --routine add
```

New people the owner confirmed: `person ensure` as `/sync-recordings` step 3
does.

**Text or a file.** First write it to
`10-inbox/` with the owner's intent on the first line:

```markdown
<!-- added with /add on <today>: "<the owner's words: what it is, for which project>" -->
```

Then dispatch **one** worker exactly as `/sync-recordings` step 6
("Extract and write") dispatches an inbox item, with the routing the
owner just confirmed (type, projects, attendees) in its briefing. Verify the
note is on disk, then move the inbox file out:

```bash
python3 "$VAULTLIB" sync archive --vault "$VAULT" --path "10-inbox/<name>"
```

No note written → leave the file in Input (the evening run retries it), say
so, release the lock and end. Then mark it accepted:

```bash
cat > "$SCRATCH/add.payload.json" <<'JSON'
[{"path": "20-meetings/<stem>.md", "action": "accept"}]
JSON
python3 "$VAULTLIB" meeting triage-apply --vault "$VAULT" --items "$SCRATCH/add.payload.json"
python3 "$VAULTLIB" sync reconcile-reviews --vault "$VAULT"
python3 "$VAULTLIB" sync rederive-entities --vault "$VAULT"
```

**This conversation.**

```bash
cat > "$SCRATCH/session.payload.json" <<'JSON'
{"title": "Pricing approach for Acme", "date": "<today>",
 "projects": ["acme-erp-rollout"], "people": ["jan-peeters"],
 "summary": ["Fixed price for phase 1", "Time and material after go-live"],
 "decisions": ["Phase 1 is fixed price"],
 "scope": "what we decided about pricing"}
JSON
python3 "$VAULTLIB" session write --vault "$VAULT" --note "$SCRATCH/session.payload.json"
```

`scope` is `"whole"` when the owner named no part. `decisions` lists the
decision candidates you saw (may be empty); section C applies the
material-decision test to them.

### 6. Process

Follow `${CLAUDE_PLUGIN_ROOT}/skills/evening-sync/references/process.md` with
the **On demand** column of its Inputs table, exactly as `/sync-recordings`
step 9 does: sections A, B and C, `<today>` from rule 6, `<owner-slug>` =
`owner` from `config read`, `sync_notes` lines to step 8's Warnings. Then
release the lock.

### 7. Decisions in the chat

As `/sync-recordings` step 10: list `decisionsWritten`, ask **"Keep or
dismiss?"**, and run `decision review --keep|--dismiss` with
`--path "<decisionsDir>/<stem>.md"`. "Not now" → they stay in Decisions › To review.

### 8. Summary

```
## Added
**What:** Meeting · Acme ERP rollout · "Forecast call moves to Thursday"  (or Session · …)
**Project pages updated:** {names}
**Decisions:** {N} kept · {N} dismissed · {N} left in Decisions › To review
**Warnings:** {verbatim} | none
```

Then sweep scratch:

```bash
python3 "$VAULTLIB" scratch sweep --dir "$SCRATCH"
```
````
